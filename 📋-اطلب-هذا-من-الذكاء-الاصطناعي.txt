```javascript
app.post('/api/v1/network/tests', (req, res) => {
  try {
    const { ssid, bssid, testType } = req.body;
    
    if (!ssid) {
      return res.status(400).json({ success: false, error: 'يرجى إدخال اسم الشبكة (SSID)' });
    }

    const testId = 'network-' + Date.now();
    const insertTest = db.prepare(`
      INSERT INTO network_test_sessions (id, user_id, ssid, bssid, status, config)
      VALUES (?, ?, ?, ?, ?, ?)
    `);
    
    insertTest.run(testId, 'user-002', ssid, bssid || '00:00:00:00:00:00', 'PENDING', JSON.stringify({ test_type: testType || 'FULL' }));

    const test = db.prepare('SELECT * FROM network_test_sessions WHERE id = ?').get(testId);
    res.json({ 
      success: true, 
      data: { 
        ...test,
        sessionId: testId
      }, 
      message: 'تم بدء اختبار الشبكة بنجاح' 
    });
  } catch (error) {
    console.error('Create network test error:', error);
    res.status(500).json({ success: false, error: 'فشل إنشاء اختبار الشبكة' });
  }
});
```

🎯 ما سيعطيك الذكاء الاصطناعي:
═══════════════════════════════════════════════════════════════

كود معدّل يقرأ البيانات بالشكل الصحيح:

```javascript
app.post('/api/v1/network/tests', (req, res) => {
  try {
    const { target, config } = req.body;
    
    // التحقق من البيانات المطلوبة
    if (!target || !target.ssid) {
      return res.status(400).json({ 
        success: false, 
        error: 'يرجى إدخال اسم الشبكة (SSID)' 
      });
    }
    
    if (!target.authorizationId) {
      return res.status(400).json({ 
        success: false, 
        error: 'يرجى اختيار تصريح نشط' 
      });
    }
    
    // التحقق من التصريح
    const authorization = db.prepare(
      'SELECT * FROM authorizations WHERE id = ? AND status = ?'
    ).get(target.authorizationId, 'ACTIVE');
    
    if (!authorization) {
      return res.status(403).json({ 
        success: false, 
        error: 'التصريح غير صالح أو منتهي' 
      });
    }
    
    // إنشاء الفحص
    const testId = 'network-' + Date.now();
    const insertTest = db.prepare(`
      INSERT INTO network_test_sessions 
      (id, user_id, authorization_id, ssid, bssid, status, config)
      VALUES (?, ?, ?, ?, ?, ?, ?)
    `);
    
    insertTest.run(
      testId, 
      'user-002',
      target.authorizationId,
      target.ssid, 
      target.bssid || '00:00:00:00:00:00', 
      'PENDING', 
      JSON.stringify({
        testTypes: config?.testTypes || ['TRAFFIC_ANALYSIS'],
        maxAttempts: config?.maxAttempts || 100
      })
    );

    const test = db.prepare(
      'SELECT * FROM network_test_sessions WHERE id = ?'
    ).get(testId);
    
    res.json({ 
      success: true, 
      data: { 
        ...test,
        sessionId: testId
      }, 
      message: 'تم بدء اختبار الشبكة بنجاح' 
    });
  } catch (error) {
    console.error('Create network test error:', error);
    res.status(500).json({ 
      success: false, 
      error: 'فشل إنشاء اختبار الشبكة' 
    });
  }
});
```
