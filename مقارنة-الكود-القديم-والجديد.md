```typescript
/**
 * استخراج الروابط من الصفحة
 */
private extractLinks($: cheerio.CheerioAPI, baseUrl: string): string[] {
  const links: string[] = [];

  // 1. استخراج روابط <a href>
  $('a[href]').each((_, element) => {
    const href = $(element).attr('href');
    if (href) {
      const absoluteUrl = this.resolveUrl(href, baseUrl);
      if (absoluteUrl && this.isSameDomain(absoluteUrl)) {
        links.push(absoluteUrl);
      }
    }
  });

  // 2. استخراج روابط من <link href> (stylesheets, etc)
  $('link[href]').each((_, element) => {
    const href = $(element).attr('href');
    if (href && !href.endsWith('.css') && !href.endsWith('.ico')) {
      const absoluteUrl = this.resolveUrl(href, baseUrl);
      if (absoluteUrl && this.isSameDomain(absoluteUrl)) {
        links.push(absoluteUrl);
      }
    }
  });

  // 3. استخراج روابط من <script src>
  $('script[src]').each((_, element) => {
    const src = $(element).attr('src');
    if (src && (src.includes('/api/') || src.includes('/rest/'))) {
      const absoluteUrl = this.resolveUrl(src, baseUrl);
      if (absoluteUrl && this.isSameDomain(absoluteUrl)) {
        links.push(absoluteUrl);
      }
    }
  });

  // 4. استخراج روابط من <img src> (قد تكون endpoints)
  $('img[src]').each((_, element) => {
    const src = $(element).attr('src');
    if (src && (src.includes('/api/') || src.includes('/rest/'))) {
      const absoluteUrl = this.resolveUrl(src, baseUrl);
      if (absoluteUrl && this.isSameDomain(absoluteUrl)) {
        links.push(absoluteUrl);
      }
    }
  });

  // 5. إضافة مسارات شائعة للمواقع الضعيفة
  const commonPaths = [
    '/login', '/admin', '/api', '/rest', '/search',
    '/profile', '/user', '/users', '/products', '/product',
    '/cart', '/checkout', '/contact', '/about', '/dashboard',
    '/settings', '/account', '/register', '/signup',
    '/forgot-password', '/reset-password',
  ];

  // إضافة المسارات الشائعة فقط إذا لم نجد روابط كافية
  if (links.length < 5) {
    const baseUrlObj = new URL(baseUrl);
    for (const path of commonPaths) {
      const testUrl = `${baseUrlObj.protocol}//${baseUrlObj.hostname}${baseUrlObj.port ? ':' + baseUrlObj.port : ''}${path}`;
      links.push(testUrl);
    }
    console.log(`   📌 Added ${commonPaths.length} common paths for better coverage`);
  }

  return [...new Set(links)]; // إزالة التكرارات
}
```

**التحسينات**:
- ✅ يبحث في 4 مصادر مختلفة (a, link, script, img)
- ✅ يكتشف API endpoints
- ✅ يضيف 20 مسار شائع تلقائياً
- ✅ يوفر تغطية أفضل للمواقع الحديثة

**النتيجة**: 10-50 صفحة تم فحصها ✅

### ✅ الكود الجديد - crawlPage():

```typescript
private async crawlPage(url: string, depth: number): Promise<CrawlResult> {
  this.visitedUrls.add(url);
  this.requestCount++;

  try {
    const response = await this.axiosInstance.get(url, {
      validateStatus: (status) => status < 500, // قبول 404, 403, etc
    });
    
    // تخطي الصفحات التي ليست HTML
    const contentType = response.headers['content-type'] || '';
    if (!contentType.includes('text/html') && !contentType.includes('application/json')) {
      console.log(`   ⏭️ Skipping non-HTML content: ${contentType}`);
      return {
        url,
        statusCode: response.status,
        contentType,
        links: [],
        forms: [],
        depth,
      };
    }

    const $ = cheerio.load(response.data);

    // استخراج الروابط
    const links = this.extractLinks($, url);

    // استخراج النماذج
    const forms = this.extractForms($, url);

    return {
      url,
      statusCode: response.status,
      contentType: response.headers['content-type'],
      links,
      forms,
      depth,
    };
  } catch (error) {
    // في حالة الخطأ، نرجع نتيجة فارغة بدلاً من رمي الخطأ
    console.log(`   ⚠️ Failed to crawl ${url}, continuing...`);
    return {
      url,
      statusCode: 0,
      contentType: undefined,
      links: [],
      forms: [],
      depth,
    };
  }
}
```

**التحسينات**:
- ✅ يقبل status codes < 500 (404, 403 مقبولة)
- ✅ يتخطى المحتوى غير HTML
- ✅ يستمر في الزحف حتى عند الأخطاء
- ✅ لا يتوقف الفحص بالكامل

### ✅ الكود الجديد (الحل):

```javascript
async handleNetworkAnalysisSubmit(e) {
  e.preventDefault();
  
  const form = e.target;
  const formData = new FormData(form);
  
  // التحقق من التصريح
  const authorizationId = formData.get('authorizationId');
  if (!authorizationId) {
    window.notificationManager.show('يجب اختيار تصريح نشط', 'error');
    return;
  }
  
  // ✅ التحقق من BSSID
  const bssid = formData.get('bssid');
  if (!bssid || !bssid.trim()) {
    window.notificationManager.show('يجب إدخال BSSID (عنوان MAC للشبكة)', 'error');
    return;
  }
  
  // ✅ الحصول على SSID (اختياري لكن يفضل وجوده)
  const ssid = formData.get('ssid');
  if (!ssid || !ssid.trim()) {
    const confirmWithoutSSID = confirm('لم تقم بإدخال SSID (اسم الشبكة). هل تريد المتابعة؟');
    if (!confirmWithoutSSID) {
      return;
    }
  }
  
  // جمع أنواع الاختبارات المحددة
  const testTypes = [];
  form.querySelectorAll('input[name="testTypes"]:checked').forEach(checkbox => {
    testTypes.push(checkbox.value);
  });

  if (testTypes.length === 0) {
    window.notificationManager.show('يجب اختيار نوع اختبار واحد على الأقل', 'error');
    return;
  }

  // ✅ الحل: authorizationId داخل target
  const data = {
    target: {
      ssid: ssid && ssid.trim() ? ssid.trim() : 'Unknown Network',
      bssid: bssid.trim(),
      authorizationId: authorizationId  // ✅ صحيح!
    },
    config: {
      testTypes: testTypes
    }
  };

  // ... إرسال البيانات
}
```

**التحسينات**:
- ✅ `authorizationId` داخل `target` (يطابق Backend)
- ✅ تحقق من BSSID قبل الإرسال
- ✅ تأكيد إذا لم يتم إدخال SSID
- ✅ تنظيف البيانات (trim)
- ✅ رسائل خطأ واضحة

**النتيجة**: 
```
✅ تم بدء تحليل الشبكة بنجاح!
✅ الانتقال إلى صفحة Network
✅ ظهور التحليل في القائمة
```
